\begin{frame}
  \frametitle{Cardy Trick}
  \framesubtitle{A cheecky way to generate boundaries}
  Therefore the partition functions are related by\par
  <Image of partition function decompositions into traces>
  \begin{equation}
    \mathrm{Tr\,}_{\mathbb{H}} e^{-2\pi L H} \alpha \beta^\dagger = Z(\tau) = \mathrm{Tr\,}_{\mathbb{H}_{\alpha \beta}}e^{2\pi i P}
  \end{equation}

\begin{frame}
  \frametitle{Cardy Trick}
  \framesubtitle{A cheecky way to generate boundaries}
  We can then project into an annulus\par
  <Map to annulus>
  \begin{equation}
    Z(\tau) = \langle \beta, q^{\frac{1}{2} \left(L_0 + \bar L_0 - \frac{c}{12} \right)} \alpha \rangle
  \end{equation}
\end{frame}

\begin{frame}
  \frametitle{Cardy Trick}
  \framesubtitle{A cheecky way to generate boundaries}
  But also an upper half plane\par
  <map to upper half plane>
  \begin{equation}
    Z(\tau) = \mathrm{Tr\,}_{\mathbb{H}_{\alpha \beta}} \tilde q^{L_0 - \frac{c}{24}}
  \end{equation}
\end{frame}


\begin{frame}
